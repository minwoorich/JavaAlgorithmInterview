import java.util.*;

class Solution {
    public String solution(int[] numbers) {
        
        String[] numbersStr = toStringArray(numbers);
        // sort() 로 정렬 기준이 될 Comparable 객체를 전
        // 기본적으로 내림차순 정렬을 진행하지만, 순서를 바꾸는게 손해인 케이스들을 커버
        // b+a > a+b 이면 swap!
        Arrays.sort(numbersStr, (a,b)-> (b+a).compareTo(a+b)); 
        
        return createToString(numbersStr);
        
    }
    
    public boolean isZero(int[] numbers){
        for(int num : numbers){
            if(num!=0){
                return false;
            }
        }
        return true;
    }
    
    public String[] toStringArray(int[] numbers){
        String[] result = new String[numbers.length];
        
        for(int i=0; i<numbers.length; i++){
            result[i] = String.valueOf(numbers[i]);
        }
        return result;
    }
    
    public String createToString(String[] arr){
        StringBuilder sb = new StringBuilder();
        for(String str : arr){
            sb.append(str);
        }
        return sb.toString();
    }
}
